import streamlit as st
import pdfplumber
import pandas as pd
import re
import io

st.set_page_config(page_title="MSDS CAS 유해물질 자동 검별기", layout="wide")
st.title("🧪 MSDS 유해물질 자동 검별 도구")
st.write("MSDS PDF 파일을 업로드하면 CAS 번호를 추출해 유해물질 목록과 자동으로 대조합니다.")

# CAS 번호 정규식 패턴 (예: 50-00-0, 1330-20-7)
CAS_REGEX = r'\b[1-9]\d{1,6}-\d{2}-\d\b'

# 1. 사이드바: 유해물질 기준 DB(엑셀) 업로드
st.sidebar.header("📁 1. 유해물질 기준 DB 등록")
db_file = st.sidebar.file_uploader("유해물질 목록 엑셀 업로드 (.xlsx)", type=["xlsx", "xls"])

target_dict = {}
if db_file:
    df_db = pd.read_excel(db_file)
    # 엑셀에 'CAS' 컬럼과 '물질명' 또는 '규제구분' 컬럼이 있다고 가정
    cas_col = [c for c in df_db.columns if 'CAS' in str(c).upper()]
    if cas_col:
        c_name = cas_col[0]
        # {CAS번호: 규제정보나 물질명} 형태로 매핑 딕셔너리 생성
        for _, row in df_db.iterrows():
            cas_val = str(row[c_name]).strip()
            # 비고/규제구분 컬럼이 있으면 함께 표시
            info = row.get('규제구분', row.get('물질명', '유해물질 해당'))
            target_dict[cas_val] = str(info)
        st.sidebar.success(f"기준 DB 등록 완료: 총 {len(target_dict)}종 로드됨")
    else:
        st.sidebar.error("엑셀 파일 내에 'CAS'라는 이름이 들어간 컬럼이 필요합니다.")

# 2. 메인 화면: 검토할 MSDS PDF 업로드 (다중 선택 가능)
st.subheader("📄 2. 검토할 MSDS PDF 파일 업로드")
uploaded_pdfs = st.file_uploader("하나 또는 여러 개의 MSDS PDF를 선택하세요", type=["pdf"], accept_multiple_files=True)

if uploaded_pdfs:
    all_results = []
    
    with st.spinner("PDF 분석 중..."):
        for pdf_file in uploaded_pdfs:
            extracted_cas = set()
            
            # PDF 텍스트 읽기
            with pdfplumber.open(pdf_file) as pdf:
                for page_num, page in enumerate(pdf.pages):
                    text = page.extract_text() or ""
                    # CAS 번호 검색
                    matches = re.findall(CAS_REGEX, text)
                    for m in matches:
                        extracted_cas.add(m)
            
            # 결과 정리
            if not extracted_cas:
                all_results.append({
                    "파일명": pdf_file.name,
                    "추출된 CAS": "미발견(이미지 스캔본 의심)",
                    "유해물질 해당 여부": "-",
                    "상세 규제 정보": "-"
                })
            else:
                for cas in sorted(extracted_cas):
                    is_hazard = cas in target_dict if target_dict else "DB 미등록"
                    hazard_info = target_dict.get(cas, "-") if target_dict else "-"
                    
                    all_results.append({
                        "파일명": pdf_file.name,
                        "추출된 CAS": cas,
                        "유해물질 해당 여부": "⚠️ 해당" if (is_hazard is True) else ("정상 (미해당)" if is_hazard is False else "확인필요"),
                        "상세 규제 정보": hazard_info
                    })

    # 3. 결과 테이블 표시
    df_results = pd.DataFrame(all_results)
    st.subheader("📊 검토 결과")
    st.dataframe(df_results, use_container_width=True)

    # 4. 엑셀 다운로드 버튼
    buffer = io.BytesIO()
    with pd.ExcelWriter(buffer, engine='openpyxl') as writer:
        df_results.to_excel(writer, index=False, sheet_name='검토결과')
    
    st.download_button(
        label="📥 결과 엑셀 파일로 다운로드",
        data=buffer.getvalue(),
        file_name="MSDS_유해물질_검토결과.xlsx",
        mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
    )
